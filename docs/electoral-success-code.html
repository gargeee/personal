#Load data
finance = pd.read_csv("candidate_summary_2022.csv")
results = pd.read_csv("election_results_house.csv")

#Adjust at large districts in finance to 1 instead of 00
finance["Cand_Office_Dist"] = pd.to_numeric(finance["Cand_Office_Dist"], errors = "coerce")
finance.loc[finance["Cand_Office_Dist"] == 0, "Cand_Office_Dist"] = 1
finance["Cand_Office_Dist"] = finance["Cand_Office_Dist"].astype("Int64")

#Filter for only 2022 U.S. House general election results
results_2022 = results[(results["cycle"] == 2022) & (results["office_name"] == "U.S. House") & (results["stage"] == "general")].copy()

#Clean up column values
results_2022["Cand_Office_St"] = results_2022["state_abbrev"]
results_2022["Cand_Office_Dist"] = (results_2022["office_seat_name"].str.extract(r"(\d+)").astype(int))

def last_name(name):
    if pd.isna(name):
        return np.nan
    name = str(name)
    if "," in name:
        last = name.split(",")[0]
    else:
        last = name.split()[-1]
    return last.strip().upper()

finance["last_name"] = finance["Cand_Name"].apply(last_name)
results_2022["last_name"] = results_2022["candidate_name"].apply(last_name)

#Create winner column
winners = results_2022[results_2022["winner"].astype(bool)][["Cand_Office_St", "Cand_Office_Dist", "last_name", "winner"]].drop_duplicates()
summary = finance.merge(winners, on = ["Cand_Office_St", "Cand_Office_Dist", "last_name"], how = "left")
summary["winner"] = summary["winner"].astype("boolean").fillna(False)
summary["win_general_2022"] = summary["winner"].astype(int)

#Filter for house elections from finance data
summary = summary[(summary["Cand_Office"] == "H") & (summary["Cand_Office_Dist"] > 0)].copy()

#Create race_Id column
summary["Cand_Office_Dist_int"] = summary["Cand_Office_Dist"].astype(int)
summary["race_id"] = (summary["Cand_Office_St"] + "-" + summary["Cand_Office_Dist_int"].astype(str))

#Create number of candidates in each race column
summary["num_candidates_in_race"] = (summary.groupby(["race_id"])["Cand_Id"].transform("count"))

#Save the new dataset
summary.to_csv("finance_and_results.csv", index = False)

#Clean up column values
def adjust(df):
    df = df.copy()
    df["Cand_Office_St"] = df["State"].map(state_to_abbrev)
    #Some districts labeled "At-Large"
    def district_adj(x):
        if pd.isna(x):
            return np.nan
        district = str(x).strip()
        if district.lower().startswith("at-large"):
            return 1
        try:
            return float(district)
        except ValueError:
            return np.nan
    df["Cand_Office_Dist"] = df["District"].apply(district_adj)
    df["endorse_name"] = df["Candidate"]
    return df

    w_lower = w.lower()
    if "latino" in w_lower:
        return "Latino"
    if "asian" in w_lower:
        return "Asian"
    if "black" in w_lower:
        return "Black"
    if "white" in w_lower:
        return "White"
    if "middle eastern" in w_lower:
        return "Middle Eastern"
    if "native american" in w_lower:
        return "Native American"
    if "pacific islander" in w_lower:
        return "Pacific Islander"
    return "Unknown"
everything["Race 1"] = everything["Race 1"].fillna('')
everything["Race 1"] = everything["Race 1"].apply(simplify_races)

#Adjusting endorsements to have binary values and imputing 0 for unknown values

from sklearn.linear_model import LinearRegression, LogisticRegression
from sklearn.preprocessing import StandardScaler
from scipy.stats import norm
import numpy as np

T = "log_Total_Receipt"
Y = "win_general_2022"
COVARIATES = ["Cand_Office_St", "Cand_Office_Dist_int", "num_candidates_in_race", "Trump", "Club for Growth", 
              "Party Committee", "Renew America", "E-PAC", "VIEW PAC", "Maggie's List", "Winning for Women", 
              "EMILY's List", "Justice Dems", "Indivisible", "PCCC", "Our Revolution", "Sunrise", 
              "Sanders", "AOC", "Unemployment", "Median Income", "Population", 
              "Cand_Incumbent_Challenger_Open_Seat_CHALLENGER", "Cand_Incumbent_Challenger_Open_Seat_INCUMBENT",
              "Cand_Incumbent_Challenger_Open_Seat_OPEN", "Cand_Party_Affiliation_DEM", "Cand_Party_Affiliation_REP",
              "Cand_Party_Affiliation_THIRD", "Gender_Female", "Gender_Male", "Gender_Nonbinary", "Race 1_Asian",
              "Race 1_Black", "Race 1_Latino", "Race 1_Middle Eastern", "Race 1_Native American", "Race 1_Pacific Islander",
              "Race 1_Unknown", "Race 1_White"]

#Handle empty values
house_ohe[COVARIATES] = house_ohe[COVARIATES].replace([np.inf, -np.inf], np.nan)

#Drop still missing values
house_ohe = house_ohe.dropna(subset=[T, Y])

#Scale confounders
scaler = StandardScaler()
X_scaled = scaler.fit_transform(house_ohe[COVARIATES])

#Form treatment model through general propensity scores
treat_model = LinearRegression()
treat_model.fit(X_scaled, house_ohe[T])

#Make predictions using treatment models for mean and standard deviation
mu_hat = treat_model.predict(X_scaled)
sigma_hat = np.std(house_ohe[T] - mu_hat)

#Add general propensity scores to data frame
house_ohe["gps"] = norm.pdf(house_ohe[T], loc=mu_hat, scale=sigma_hat)

#Stabilize propensity scores and add to data frame
m_mu = house_ohe[T].mean()
m_sd = house_ohe[T].std()
house_ohe["gps_w_stab"] = (norm.pdf(house_ohe[T], m_mu, m_sd)/ house_ohe["gps"])

#Weighted outcome model
outcome_model = LogisticRegression(max_iter=1000)
outcome_model.fit(house_ohe[[T]], house_ohe[Y], sample_weight=house_ohe["gps_w_stab"])
