#Data Cleaning

#Dropped rows with missing values in any of the selected columns
df = df.dropna()
#Sanity checking the amount of data left after dropping columns with missing values
#In our case, we deemed 238,739 was sufficient to proceed
print("After dropna shape:", df.shape)

#Set proper data types for severity, weather, and sunrise/sunset columns
df["Severity"] = df["Severity"].astype(int)
df["Weather_Condition"] = df["Weather_Condition"].astype(str)
df["Sunrise_Sunset"] = df["Sunrise_Sunset"].astype(str)

#Defined a function to simplify weather into our own 7 categories for simplicity using synonymous conditions
def simplify_weather(w):
    w_lower = w.lower()
    if "snow" in w_lower or "sleet" in w_lower or "ice" in w_lower or "wintry" in w_lower:
        return "Snow/Ice"
    if "rain" in w_lower or "drizzle" in w_lower or "shower" in w_lower:
        return "Rain"
    if "storm" in w_lower or "thunder" in w_lower or "tstorm" in w_lower:
        return "Storm"
    if "fog" in w_lower or "mist" in w_lower or "haze" in w_lower or "smoke" in w_lower:
        return "Fog/Mist"
    if "cloud" in w_lower or "overcast" in w_lower:
        return "Cloudy"
    if "clear" in w_lower or "fair" in w_lower:
        return "Clear"
    return "Other"

#Applied this function to the existing column in a new column
df["Weather_Condition_Simple"] = df["Weather_Condition"].apply(simplify_weather)

#Checking the distribution of severity classification within the dataset
print("Severity value counts:")
print(df["Severity"].value_counts())

#Checking the distribution of weather classification within the dataset using our simplified column
print("\nSimplified weather counts:")
print(df["Weather_Condition_Simple"].value_counts().head())

#Data preprocessing

#All numerical features that are already in int/float format
num_features = ["Start_Lat", "Start_Lng", "Distance(mi)", "Temperature(F)", "Humidity(%)", "Pressure(in)", "Visibility(mi)",
                "Wind_Speed(mph)"]
#All categorical features
cat_features = ["Weather_Condition_Simple", "Sunrise_Sunset"]

#Using StandardScaler to preprocess numerical data and one hot encoding categorical data
preprocess = ColumnTransformer(
    transformers=[('num', StandardScaler(), num_features), ('cat', OneHotEncoder(handle_unknown='ignore'), cat_features)])

#Fit transformer only on training data
X_train_processed = preprocess.fit_transform(X_train)
X_test_processed = preprocess.transform(X_test)
print("Shape after preprocessing (train):", X_train_processed.shape)

#Apply SMOTE to handle class imbalance on the training set
smote = SMOTE(random_state=42)
X_train_bal, y_train_bal = smote.fit_resample(X_train_processed, y_train)
#Ensure that SMOTE was applied successfully and class counts have changed since before/after applying SMOTE
print("Class counts before SMOTE:", np.bincount(y_train))
print("Class counts after SMOTE: ", np.bincount(y_train_bal))
